VARIABLE DESCRIPTIONS
═══════════════════════════════════════════════════════════════════════════════

NODE_ENV=production
  ↳ Sets environment to production mode

PORT=4000
  ↳ Backend runs on port 4000

MONGODB_URI=mongodb+srv://...
  ↳ Connection string to MongoDB Atlas database

JWT_SECRET=YWVjMjZhYjQt...
  ↳ Secret key for JWT token signing (randomly generated)

JWT_EXPIRES_IN=7d
  ↳ JWT tokens expire after 7 days

CORS_ORIGIN=https://codingcon-frontend.vercel.app
  ↳ Allow requests from your Vercel frontend

CLUSTER_ENABLED=true
  ↳ Enable Node.js clustering for multi-core CPU

WORKERS=4
  ↳ Spawn 4 worker processes (handles ~500 users)

DB_POOL_SIZE=50
  ↳ Maximum MongoDB connections in pool

DB_MIN_POOL_SIZE=10
  ↳ Minimum MongoDB connections in pool

MAX_JUDGE_JOBS=4
  ↳ Maximum concurrent code judgments

QUEUE_CONCURRENCY=2
  ↳ Maximum concurrent queue workers

CRON_SECRET=YmUxOGFlZDMt...
  ↳ Secret key for scheduled tasks (randomly generated)
